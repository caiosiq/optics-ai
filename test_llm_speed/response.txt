import numpy as np

def write_cavity_design():
    """
    Calculate geometrical parameters and design for a stable Nd:YAG laser cavity
    pumped at 250 mW, 1% doping, 5 mm thickness crystal with given optics.
    Outputs parameters unambiguously in a text file.
    """

    # Constants
    wavelength = 1064e-9         # 1064 nm (Nd:YAG fundamental) in meters
    crystal_thickness = 5e-3     # 5 mm in meters
    pump_power = 250e-3          # 250 mW in W
    pump_radius = 0.5e-3         # 0.5 mm 1/e² pump radius at input face in meters
    lens_f = 0.1                 # Lens focal length: 0.1 m
    input_coupler_R = 0.99       # Input coupler reflectivity (flat)
    output_coupler_R = 0.9974    # Output coupler reflectivity
    output_coupler_ROC = 0.15    # 15 cm radius of curvature in meters
    n_crystal = 1.82             # Refractive index Nd:YAG at 1064 nm (approx)
    crystal_pos = 0              # define crystal start at z=0
    
    # --- Cavity Design ---
    # Proposal: Linear cavity: Flat input coupler (IC) mirror — crystal — lens — curved output coupler (OC)
    # Order: IC (flat) @ z=0, crystal 5mm, lens somewhere after crystal, OC ROC=15cm, 
    # Distances chosen to maintain cavity stability and maximize overlap
    
    # Step 1: Define cavity elements positions along optical axis (z)
    # Choose lens right after crystal at z = crystal_thickness + air gap
    air_gap_after_crystal = 0.02    # 2 cm air gap after crystal before lens
    lens_pos = crystal_thickness + air_gap_after_crystal  # lens position
    
    # Output coupler position after lens at distance 'L_oc' to be tuned for stability
    # For a stable cavity, stability condition: 0 <= g1*g2 <= 1
    # g1 = 1 - L/R1 , g2 = 1 - L/R2 (R1,R2 radii of curvature mirrors)
    # Here, input coupler is flat => R1 = inf => g1 = 1 - L/inf = 1
    # output coupler R2 = 0.15 m (concave)
    # L = total cavity length from mirror 1 to mirror 2 (IC to OC)
    # We want g1*g2 in (0,1), so 0 <= (1)*(1 - L/0.15) <=1 => 0<=1-L/0.15<=1 => 0<=1-L/0.15<=1
    # => L<=0.15m
    
    # So cavity length from IC to OC must be <15cm to be stable.
    # But lens + crystal + gap + output coupler distance must also satisfy this:
    # L = crystal_thickness + air_gap_after_crystal + lens_to_OC_distance
    # Lens is about focal length f=0.1m, cavity length ~0.13m is reasonable.
    
    # Let lens_to_OC_distance = L_oc_dist
    L_oc_dist = 0.1                 # 10 cm from lens to output coupler
    
    cavity_length = crystal_thickness + air_gap_after_crystal + L_oc_dist
    
    # Calculate g parameters
    R1 = np.inf    # flat input coupler
    R2 = output_coupler_ROC

    L = cavity_length
    g1 = 1.0                  # flat mirror
    g2 = 1.0 - L / R2

    stability = g1 * g2
    stability_margin = min(stability, 1 - stability)

    # Step 2: Calculate beam parameters
    # Assume ABCD matrix from IC to OC including crystal and lens elements.
    # Simplify: Treat crystal as medium of length 5mm and n=1.82 => optical length inside crystal = n*L
    # Air gaps propagate differently.
    #
    # We treat the cavity as:
    # IC (plane) at z=0
    # Crystal (n=1.82, L=5mm)
    # Air 2 cm
    # Lens f=0.1 m (thin lens)
    # Air L_oc_dist (10 cm)
    # OC curvature mirror
    
    # Build ABCD matrices for one round trip from IC to OC and back (or single pass propagation)
    #
    # Using complex beam parameter method for cavity mode:
    # Find q parameter at IC that satisfies q = (A*q + B)/(C*q + D)
    # q = 1/(1/R - j λ/(π w²)), want to find beam waist at crystal center
    
    # Define matrices (in proper order colinear propagation):
    # Propagation in medium length L and refractive index n:
    # free space propagation matrix M_free = [[1, L], [0,1]]
    # But inside crystal refractive index modifies wavelength λ/n to λ'
    
    # Optical path length in crystal is n*L physically, but ABCD matrix uses physical length and n only indirectly.
    # For Gaussian beams in media, wavelength inside medium = λ / n
    # Propagation matrices depend on physical distance only
    
    # Define matrices:
    def M_propagation(L):
        return np.array([[1, L],
                         [0, 1]])
    def M_lens(f):
        return np.array([[1, 0],
                         [-1/f, 1]])
    
    # Propagation:
    M_crystal = M_propagation(crystal_thickness)
    M_air1 = M_propagation(air_gap_after_crystal)
    M_air2 = M_propagation(L_oc_dist)
    M_lens = M_lens(lens_f)
    
    # Total ABCD matrix from IC to OC:
    # M_total = M_air2 * M_lens * M_air1 * M_crystal
    # Matrix multiplication order: right to left (first propagate crystal, then air1, then lens, then air2)

    M_total = M_air2 @ M_lens @ M_air1 @ M_crystal

    A = M_total[0,0]
    B = M_total[0,1]
    C = M_total[1,0]
    D = M_total[1,1]

    # Mirror curvatures IC: flat (R=inf), OC: R=0.15 m
    # At IC mirror, radius of curvature = inf, q parameter to find such that beam reproduces at each round trip

    # Eigenmode q satisfies: q = (A*q + B)/(C*q+D)

    # Rewrite as quadratic in 1/q:
    # Actually better to solve:
    # q = (A*q + B) / (C*q + D)
    # => (C*q + D)*q = A*q + B
    # => C q^2 + D q = A q + B
    # => C q^2 + D q - A q - B = 0
    # => C q^2 + (D - A) q - B = 0
    
    # Let q be complex with positive imaginary part

    # Solve quadratic:
    coeffs = [C, (D - A), -B]
    roots = np.roots(coeffs)

    # Choose root with positive imaginary part (physical beam)
    q_mode = None
    for r in roots:
        if np.imag(r) > 0:
            q_mode = r
            break
    if q_mode is None:
        q_mode = roots[0]  # fallback

    # Beam waist size w0 and location z0 can be derived from q parameter:
    # q = z + j z_R
    # 1/q = 1/R - j λ/(pi w^2)
    # lambda inside crystal = wavelength/n_crystal
    # We'll calculate w inside crystal

    # Calculate beam radius and curvature at crystal input interface (z=0)
    q_inv = 1/q_mode
    R_cavity = 1/np.real(q_inv) if np.real(q_inv) != 0 else np.inf
    w_cavity = np.sqrt(-wavelength/(np.pi * np.imag(q_inv)))  # in air approximation
    
    # For crystal medium, effective wavelength = wavelength / n_crystal
    w_cavity_crystal = np.sqrt(- (wavelength / n_crystal) / (np.pi * np.imag(q_inv)))

    # Beam waist location relative to IC mirror:
    # q = z + j z_R, z_R = Rayleigh length
    z_R = np.imag(q_mode)
    z_waist = -np.real(q_mode)   # distance from IC mirror to waist (if negative, waist before IC)

    # Approximate waist position inside cavity length
    waist_pos_in_cavity = z_waist
    waist_size = np.sqrt(wavelength/(np.pi * z_R))

    # Since pump beam radius is 0.5mm, and crystal thickness 5mm, check mode volume overlap
    # Mode volume roughly proportional to w² * crystal length
    mode_radius = w_cavity_crystal
    mode_volume = np.pi * mode_radius**2 * crystal_thickness
    pump_volume = np.pi * pump_radius**2 * crystal_thickness
    overlap = min(mode_volume, pump_volume) / max(mode_volume, pump_volume)  # normalized volume overlap

    # Diffraction losses (approximate): related to clipping of mode on mirror aperture
    # Assuming mirrors large enough, diffraction losses low; estimate <0.1%
    diffraction_loss = 0.001

    # Compose output unambiguously:
    lines = [
        "Cavity Design Parameters for Nd:YAG Laser:",
        "----------------------------------------",
        f"Wavelength (m): {wavelength:.3e}",
        f"Crystal: Nd:YAG, 1% doping, thickness (m): {crystal_thickness:.3e}, index: {n_crystal}",
        f"Pump Power (W): {pump_power:.3e}",
        f"Pump beam radius at crystal input face (m): {pump_radius:.3e}",
        "",
        "Optical Elements:",
        f"Input Coupler: Flat mirror (R=inf), Reflectivity: {input_coupler_R*100:.2f}%",
        f"Output Coupler: Concave mirror, ROC: {output_coupler_ROC*100:.1f} cm, Reflectivity: {output_coupler_R*100:.2f}%",
        f"Lens: Thin lens, focal length: {lens_f*100:.1f} cm",
        "",
        "Element Positions (along beam axis z, meters):",
        f"IC mirror (flat input coupler): z = 0",
        f"Crystal input face: z = 0 (coincident with IC)",
        f"Crystal thickness: {crystal_thickness*1000:.1f} mm",
        f"Air gap after crystal before lens: {air_gap_after_crystal*1000:.1f} mm",
        f"Lens position: {lens_pos*1000:.1f} mm",
        f"Output coupler position: {cavity_length*1000:.1f} mm (after lens)",
        "",
        "Cavity Length (IC to OC): {:.1f} mm".format(cavity_length*1000),
        f"Stability parameters: g1 = {g1:.3f} (flat), g2 = {g2:.3f} (ROC={output_coupler_ROC*100:.1f} cm)",
        f"Stability condition (g1*g2): {stability:.3f} (must be 0 <= value <= 1 for stable cavity)",
        f"Stability margin = min(g1*g2, 1 - g1*g2): {stability_margin:.3f}",
        "",
        f"Mode Beam Waist (radius) in crystal (m): {mode_radius:.3e}",
        f"Beam waist location from IC mirror (m): {waist_pos_in_cavity:.3e}",
        f"Beam waist Rayleigh range (m): {z_R:.3e}",
        f"Pump mode volume overlap (normalized): {overlap:.3f}",
        f"Estimated diffraction losses (fraction): {diffraction_loss:.4f}",
        "",
        "Summary:",
        f"- Cavity formed by flat IC mirror at z=0 and curved OC mirror (ROC=15 cm) at z={cavity_length*1000:.1f} mm",
        f"- Medium: 5 mm Nd:YAG crystal at IC, followed by 2 cm air gap, then 10 cm lens to OC",
        f"- Lens (f = 10 cm) positioned at 22 mm from IC (5mm crystal + 20mm air gap)",
        f"- Stable resonator with stability parameter near {stability:.3f}",
        f"- Beam waist approx {mode_radius*1e6:.1f} microns within crystal, well matched to pump spot size",
        f"- Mode overlap and low diffraction loss maximize efficiency",
    ]

    with open("ndyag_laser_cavity_design.txt", "w") as f:
        for line in lines:
            f.write(line + "\n")

if __name__ == "__main__":
    write_cavity_design()
